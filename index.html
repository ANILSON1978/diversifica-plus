function updateSimulation() { const P = parseFloat(document.getElementById('initialAmount').value) || 0; const PMT = parseFloat(document.getElementById('monthlyAmount').value) || 0; const months = parseInt(document.getElementById('months').value) || 12; const cdi = (parseFloat(document.getElementById('cdi').value) || 0) / 100; const startDateVal = document.getElementById('startDate').value; const totalInvested = P + (PMT * months); const days = months * 30; const irRate = getIRRate(days); const activeAssets = Array.from(document.querySelectorAll('.asset-checkbox:checked')).map(cb => cb.value); let investments = []; if (activeAssets.includes('cdb')) { const rate = (parseFloat(document.getElementById('cdbRate').value) || 0) / 100 * cdi; const res = calculateCompoundInterest(P, PMT, rate, months); investments.push({ id: 'cdb', name: 'CDB / LC', gross: res.finalGross, taxRate: irRate, coupons: 0, history: res.history, color: '#10b981' }); } if (activeAssets.includes('lca')) { const rate = (parseFloat(document.getElementById('lcaRate').value) || 0) / 100 * cdi; const res = calculateCompoundInterest(P, PMT, rate, months); investments.push({ id: 'lca', name: 'LCA / LCI (Pós)', gross: res.finalGross, taxRate: 0, coupons: 0, history: res.history, color: '#3b82f6' }); } if (activeAssets.includes('lcaPre')) { const rate = (parseFloat(document.getElementById('lcaPreRate').value) || 0) / 100; const res = calculateCompoundInterest(P, PMT, rate, months); investments.push({ id: 'lcaPre', name: 'LCA / LCI (Prefixada)', gross: res.finalGross, taxRate: 0, coupons: 0, history: res.history, color: '#06b6d4' }); } if (activeAssets.includes('debPre')) { const rate = (parseFloat(document.getElementById('debPreRate').value) || 0) / 100; const res = calculateCouponInterest(P, PMT, rate, months); investments.push({ id: 'debPre', name: 'Crédito Privado Pré (Cupons)', gross: res.finalGross, taxRate: 0, customTax: res.totalTaxPaid, coupons: res.totalCouponsPaid, history: res.history, color: '#e11d48' }); } if (activeAssets.includes('debPos')) { const spread = (parseFloat(document.getElementById('debPosRate').value) || 0) / 100; const totalRate = (1 + cdi) * (1 + spread) - 1; const res = calculateCouponInterest(P, PMT, totalRate, months); investments.push({ id: 'debPos', name: 'Crédito Privado Pós (Cupons)', gross: res.finalGross, taxRate: 0, customTax: res.totalTaxPaid, coupons: res.totalCouponsPaid, history: res.history, color: '#f43f5e' }); } if (activeAssets.includes('poupanca')) { const rate = (parseFloat(document.getElementById('poupancaRate').value) || 0) / 100; const res = calculateCompoundInterest(P, PMT, rate, months); investments.push({ id: 'poupanca', name: 'Poupança', gross: res.finalGross, taxRate: 0, coupons: 0, history: res.history, color: '#f59e0b' }); } if (activeAssets.includes('fundo')) { const rateCDI = (parseFloat(document.getElementById('fundoRate').value) || 0) / 100 * cdi; const fee = (parseFloat(document.getElementById('fundoFee').value) || 0) / 100; const rate = Math.max(0, rateCDI - fee); const res = calculateCompoundInterest(P, PMT, rate, months); investments.push({ id: 'fundo', name: 'Fundo Renda Fixa', gross: res.finalGross, taxRate: irRate, coupons: 0, history: res.history, color: '#8b5cf6' }); } const results = investments.map(inv => { const profitGross = (inv.gross + inv.coupons) - totalInvested; const tax = inv.customTax !== undefined ? inv.customTax : (profitGross > 0 ? profitGross * inv.taxRate : 0); const net = (inv.gross + inv.coupons) - (inv.customTax !== undefined ? 0 : tax); const netProfit = net - totalInvested; const netReturnPct = (netProfit / totalInvested) * 100; return { ...inv, totalInvested, tax, net, netProfit, netReturnPct }; }); // Cards Render const cardsContainer = document.getElementById('cardsContainer'); cardsContainer.innerHTML = ''; const sortedResults = [...results].sort((a, b) => b.net - a.net); results.forEach(res => { const isBest = sortedResults.length > 0 && res.name === sortedResults[0].name; const cardHtml = `
${isBest ? 'Maior Retorno' : ''}

${res.name}

${formatCurrency(res.net)}
Lucro Líquido: ${formatCurrency(res.netProfit)} Retorno: ${formatPercent(res.netReturnPct)}
${res.coupons > 0 ? `
Cupons Líquidos Sacados: ${formatCurrency(res.coupons)}
` : ''}
`; cardsContainer.innerHTML += cardHtml; }); // Parecer Dinâmico Render if (sortedResults.length > 0) { const melhor = sortedResults[0]; document.getElementById('parecerMelhorOpcao').textContent = melhor.name; document.getElementById('parecerValorFinal').textContent = formatCurrency(melhor.net); document.getElementById('parecerRecomendada').textContent = melhor.name; } // Tabela Render const tableBody = document.getElementById('resultsTable'); tableBody.innerHTML = ''; results.forEach(res => { const row = ` ${res.name} ${formatCurrency(res.totalInvested)} ${formatCurrency(res.gross)} ${formatCurrency(res.tax)} ${res.coupons > 0 ? formatCurrency(res.coupons) : '-'} ${formatCurrency(res.net)} ${formatPercent(res.netReturnPct)} `; tableBody.innerHTML += row; }); renderChart(months, results, startDateVal); } function renderChart(months, results, startDateStr) { const ctx = document.getElementById('growthChart').getContext('2d'); // Correção na data para evitar pulos de mês no dia 31 const baseDate = startDateStr ? new Date(startDateStr + 'T00:00:00') : new Date(); const labels = Array.from({ length: months + 1 }, (_, i) => { const d = new Date(baseDate); d.setDate(1); // Normaliza para o dia 1 d.setMonth(d.getMonth() + i); return d.toLocaleDateString('pt-BR', { month: 'short', year: '2-digit' }); }); const initialP = parseFloat(document.getElementById('initialAmount').value) || 0; const monthlyPMT = parseFloat(document.getElementById('monthlyAmount').value) || 0; const datasets = results.map(res => { const netHistory = res.history.map((val, idx) => { if (idx === 0) return val; // Se for ativo de cupom, o IR já foi descontado na fonte no momento do cupom if (res.customTax !== undefined) { return val; } // Para demais ativos, calcula o IR sobre o lucro acumulado até o mês atual (idx) const investedUntilNow = initialP + (monthlyPMT * idx); const grossProfit = val - investedUntilNow; const currentDays = idx * 30; const currentIRRate = getIRRate(currentDays); const tax = grossProfit > 0 ? grossProfit * currentIRRate : 0; return val - tax; }); return { label: res.name, data: netHistory, borderColor: res.color, backgroundColor: res.color + '1A', borderWidth: 2, tension: 0.3, pointRadius: 0 }; }); if (growthChart) growthChart.destroy(); growthChart = new Chart(ctx, { type: 'line', data: { labels, datasets }, options: { responsive: true, maintainAspectRatio: false, interaction: { mode: 'index', intersect: false }, plugins: { legend: { labels: { color: '#94a3b8' } }, tooltip: { callbacks: { label: function(context) { return `${context.dataset.label}: ${formatCurrency(context.raw)}`; } } } }, scales: { x: { grid: { color: '#334155' }, ticks: { color: '#94a3b8' } }, y: { grid: { color: '#334155' }, ticks: { color: '#94a3b8', callback: value => 'R$ ' + value.toLocaleString('pt-BR') } } } } }); }
